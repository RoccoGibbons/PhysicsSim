#version 330 core

layout (location = 0) in vec3 aPos;
layout (location = 1) in vec3 aColour;

out vec3 ourColour;
out vec4 centrePos;
out vec4 u_radius;

uniform mat4 model;
uniform mat4 view;
uniform mat4 projection;
uniform vec3 centre;
uniform float radius;

void main() {
    gl_Position = projection * view * model * vec4(aPos, 1.0f);
    ourColour = aColour;
    centrePos = projection * view * model * vec4(centre, 1.0f);
    u_radius = projection * view * model * vec4(radius, 0.0f, 0.0f, 0.0f);
}
